## 규칙
1. 모든 PK는 `BIGINT GENERATED BY DEFAULT AS IDENTITY`를 사용한다.
2. Java 매핑 시 `Long` 타입을 사용한다.
3. 모든 테이블은 `BaseEntity`의 공통 컬럼(created_at, updated_at, is_active)을 포함한다.
4. 식별자에는 ID/번호, 분류 목적의 코드값에는 코드 명명규칙을 따른다.

## 코드 관리 방침
- **Enum 관리** (Java 코드 레벨): attachment_type, role 등 고정된 시스템 분류값
- **코드 관리 필요** (추후 common_codes 테이블로 전환 검토): preferred_* 등 기획에서 선택지가 정해진 값
- **텍스트 저장** (자유입력): 흥미/적성, 기타 학습 방법 등 AI 분석용 또는 사용자 커스텀 값

## DDL 명세

```sql
-- 1. Users (회원)
CREATE TABLE users (
    users_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email       VARCHAR(255) NOT NULL UNIQUE,
    password    VARCHAR(255) NOT NULL,
    nickname    VARCHAR(100),
    age_group   VARCHAR(20),
    school_name VARCHAR(100),
    google_oauth_id VARCHAR(255),
    email_verified  BOOLEAN DEFAULT FALSE NOT NULL,
    is_admin    BOOLEAN DEFAULT FALSE NOT NULL,
    is_guest    BOOLEAN DEFAULT FALSE NOT NULL,
    role        VARCHAR(20) DEFAULT 'ROLE_USER' NOT NULL,  -- Enum: ROLE_USER, ROLE_ADMIN
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 2. Email Verifications (이메일 인증)
CREATE TABLE email_verifications (
    email_verification_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email       VARCHAR(255) NOT NULL,
    token       VARCHAR(255) NOT NULL UNIQUE,
    expires_at  TIMESTAMPTZ NOT NULL,
    verified    BOOLEAN DEFAULT FALSE NOT NULL,
    verified_at TIMESTAMPTZ,
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 3. User Settings (나만의 학습 설정)
CREATE TABLE user_settings (
    us_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id    BIGINT NOT NULL REFERENCES users(users_id),
    preferred_study_time    VARCHAR(50),    -- 코드 관리 필요 (선호 학습 시간대)
    preferred_material_format VARCHAR(100), -- 코드 관리 필요 (선호 자료 형식)
    preferred_class_style   VARCHAR(100),   -- 코드 관리 필요 (선호 수업 방식)
    preferred_study_method  VARCHAR(100),   -- 코드 관리 필요 (선호 학습 방법)
    preferred_teacher_style VARCHAR(50),    -- 코드 관리 필요 (선호 선생님 스타일)
    preferred_friend_style  VARCHAR(50),    -- 코드 관리 필요 (선호 친구 스타일)
    extra_notes             TEXT,           -- 텍스트 저장 (흥미/적성/기타 자유입력 - AI 분석용)
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 4. User Learning Styles (학습 유형 분석)
CREATE TABLE user_learning_styles (
    uls_id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id            BIGINT NOT NULL REFERENCES users(users_id),
    learning_style      VARCHAR(100),
    recommended_method  VARCHAR(255),
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active           BOOLEAN DEFAULT TRUE NOT NULL
);

-- 5. Attachments (첨부파일 - 구 notes + note_materials 통합)
-- lg_id는 학습목표 생성 전 업로드 시 NULL, 이후 연결
CREATE TABLE attachments (
    attachment_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id        BIGINT NOT NULL REFERENCES users(users_id),
    lg_id           BIGINT REFERENCES learning_goals(lg_id),  -- nullable: 생성 전 업로드 가능
    attachment_type VARCHAR(20) NOT NULL,  -- Enum: FILE, LECTURE, VIDEO, WEB, TEXT
    title           VARCHAR(255),
    file_name       VARCHAR(255),          -- FILE, LECTURE 타입
    file_url        VARCHAR(500),          -- FILE, LECTURE, VIDEO, WEB 타입
    content         TEXT,                  -- TEXT 타입 직접 입력
    extracted_text  TEXT,                  -- FILE, LECTURE 타입 - AI 분석용 추출 텍스트
    created_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active       BOOLEAN DEFAULT TRUE NOT NULL
);

-- 6. Learning Goals (학습 목표)
-- study_method 제거 → learning_goal_methods(1:N)로 분리
-- AI 분석 컬럼 제거 (difficulty_analysis, trend_analysis, recommended_method, formative_assessment)
CREATE TABLE learning_goals (
    lg_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id    BIGINT NOT NULL REFERENCES users(users_id),
    title       VARCHAR(255) NOT NULL,
    purpose     VARCHAR(100),
    target_score INTEGER,
    max_score   INTEGER,
    start_date  DATE,
    end_date    DATE,
    ai_study_plan TEXT,  -- AI 학습 계획 JSON 저장
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 7. Learning Goal Methods (학습 방법 - 1:N)
CREATE TABLE learning_goal_methods (
    lgm_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    lg_id       BIGINT NOT NULL REFERENCES learning_goals(lg_id),
    method_name VARCHAR(100) NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 8. ByDay Study Times (요일별 공부 가능 시간 - 구 daily_study_times 대체)
-- study_date(특정날짜) → day_of_week(요일 정수)로 변경
CREATE TABLE byday_study_times (
    bst_id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id        BIGINT NOT NULL REFERENCES users(users_id),
    lg_id           BIGINT NOT NULL REFERENCES learning_goals(lg_id),
    day_of_week     INTEGER NOT NULL,  -- 1=월 / 2=화 / 3=수 / 4=목 / 5=금 / 6=토 / 7=일
    available_minutes INTEGER,
    created_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active       BOOLEAN DEFAULT TRUE NOT NULL
);

-- 9. Learning Goal Excludes (학습 목표별 제외 날짜)
-- 특정 날짜(공휴일, 시험 당일 등)를 AI 학습 계획 생성에서 제외
CREATE TABLE learning_goal_excludes (
    lge_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    lg_id       BIGINT NOT NULL REFERENCES learning_goals(lg_id),
    exclude_date DATE NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active   BOOLEAN DEFAULT TRUE NOT NULL
);

-- 10. Consolidations (단권화)
-- 학습 목표 1:N 단권화 구조 (lg_id로 연결). FK 제약 없이 논리적 연관관계만 유지
CREATE TABLE consolidations (
    co_id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    users_id            BIGINT NOT NULL,
    lg_id               BIGINT NOT NULL,
    title               VARCHAR(255) NOT NULL,
    article             TEXT,           -- AI 생성 단권화 본문 (사용자 수정 가능)
    mode                VARCHAR(100),   -- 코드 관리 필요 (생성 모드)
    duplicate_handling  VARCHAR(50),    -- 코드 관리 필요 (중복 처리 방식)
    importance_level    VARCHAR(50),    -- 코드 관리 필요 (중요도)
    summary_note        TEXT,
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active           BOOLEAN DEFAULT TRUE NOT NULL
);

-- 11. Consolidation Attachments (단권화 ↔ 첨부파일 N:M 매핑, 복합키)
-- 첨부파일은 학습 목표 등록 시 이미 업로드된 attachments 중 선택되어 매핑됨
CREATE TABLE consolidation_attachments (
    co_id               BIGINT NOT NULL,
    attachment_id       BIGINT NOT NULL,
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active           BOOLEAN DEFAULT TRUE NOT NULL,
    CONSTRAINT PK_CONSOLIDATION_ATTACHMENTS PRIMARY KEY (co_id, attachment_id)
);

-- 12. Consolidation Materials (단권화 하위 자료)
CREATE TABLE consolidation_materials (
    cm_id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    co_id               BIGINT NOT NULL,
    field               TEXT NOT NULL,
    sort_order          INTEGER NOT NULL DEFAULT 0,
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_active           BOOLEAN DEFAULT TRUE NOT NULL
);
```

## 삭제된 테이블 (리팩토링)
| 테이블 | 사유 |
|---|---|
| notes | attachments로 통합 |
| note_materials | attachments로 통합 |
| memorization_plans | 기획 제거 (암기 계획 기능 삭제) |
| retrieval_plans | 기획 제거 (인출 계획 기능 삭제) |
| study_plans | 기획 제거 (기타 학습 계획 기능 삭제) |
| daily_study_times | byday_study_times로 대체 (특정날짜 → 요일별) |
| consolidation_items | consolidation_materials(field + sort_order)로 재통합 |

## 코드 관리 대상 요약
| 필드 | 테이블 | 방식 | 이유 |
|---|---|---|---|
| role | users | Enum | 시스템 고정값 |
| attachment_type | attachments | Enum | 시스템 고정값 (FILE/LECTURE/VIDEO/WEB/TEXT) |
| preferred_* | user_settings | 코드 관리 필요 | 기능 연계 |
| extra_notes | user_settings | 텍스트 저장 | AI 분석용 자유입력 |
| mode | consolidations | 코드 관리 필요 | 기능 연계 |
| duplicate_handling | consolidations | 코드 관리 필요 | 기능 연계 |
| importance_level | consolidations | 코드 관리 필요 | 기능 연계 |
